268:function matchesStudent(s: Student, lowerQuery: string): boolean {
269:  const firstName = (s.name || s.fullName || '').split(' ')[0].toLowerCase();
270:  if (!firstName || firstName.length < 2) return false;
272:  // 1. Exact Latin boundary match
273:  const latinRegex = new RegExp(`\\b${firstName}\\b`, 'i');
274:  if (latinRegex.test(lowerQuery)) return true;
276:  // 2. Multilingual aliases check (Devanagari, Bengali, Gurmukhi, Gujarati, Tamil, Telugu)
277:  const aliases = INDIC_NAME_ALIASES[firstName];
278:  if (aliases && aliases.some((alias) => lowerQuery.includes(alias.toLowerCase()))) {
279:    return true;
280:  }
282:  // 3. Fallback: Substring check if word length >= 3
283:  if (firstName.length >= 3 && lowerQuery.includes(firstName)) {
284:    return true;
285:  }
287:  return false;
288:}

734:  ): Promise<StructuredQueryAnalysis> {
